// To create a deployment
kubectl create -f deployment-definition.yaml


// To get deployments
kubectl get deployments


// to update image
kubectl set image deployment/myapp-deployment nginx=nginx:1.12.1

// to get status 
kubectl rollout status deployment/myapp-deployment

kubectl rollout history deployment/myapp-deployment


// rollback
kubectl rollout undo deployment/myapp-deployment


// To generate a yaml file through command line tool with replicas
kubectl create deployment --image=nginx nginx --replicas=5 --dry-run=client -o yaml > file-name.yaml
